  S' ::= <translation-unit> $
  
  1.) <translation-unit> ::= {<external-declaration>}*

  2.) <external-declaration> ::= <function-definition>     
                          | <declaration>                  

  4.) <declaration-specifier> ::= <storage-class-specifier>
                            | <type-specifier>
                            | <type-qualifier>

  5.) <storage-class-specifier> ::= static
                              | extern

  6.) <type-specifier> ::= void
                    | char
                    | short
                    | int
                    | long
                    | float
                    | double
                    | <struct>

  7.) <struct> ::= struct <identifier> { {<struct-declaration>}+ }
            | struct { {<struct-declaration>}+ }
            | struct <identifier>

  8.) <struct-declaration> ::= {<specifier-qualifier>}* <struct-declarator-list>

  9.) <specifier-qualifier> ::= <type-specifier>
                          | <type-qualifier>

  10.) <struct-declarator-list> ::= <declarator>
                            | <struct-declarator-list> , <declarator>

  11.) <declarator> ::= {<pointer>}? <direct-declarator>

  12.) <pointer> ::= * {<type-qualifier>}* {<pointer>}?

  13.) <type-qualifier> ::= const

  14.) <direct-declarator> ::= <identifier>
                        | ( <declarator> )
                        | <direct-declarator> [ {<constant-expression>}? ]
                        | <direct-declarator> ( <parameter-type-list> )

  15.) <constant-expression> ::= <logical-or-expression>

  16.) <logical-or-expression> ::= <logical-and-expression>
                            | <logical-or-expression> || <logical-and-expression>

  17.) <logical-and-expression> ::= <inclusive-or-expression>
                            | <logical-and-expression> && <inclusive-or-expression>

  18.) <equality-expression> ::= <relational-expression>
                          | <equality-expression> == <relational-expression>
                          | <equality-expression> != <relational-expression>

  19.) <relational-expression> ::= <additive-expression>
                            | <relational-expression> < <additive-expression>
                            | <relational-expression> > <additive-expression>
                            | <relational-expression> <= <additive-expression>
                            | <relational-expression> >= <additive-expression>

  20.) <additive-expression> ::= <multiplicative-expression>
                          | <additive-expression> + <multiplicative-expression>
                          | <additive-expression> - <multiplicative-expression>

  21.) <multiplicative-expression> ::= <cast-expression>
                                | <multiplicative-expression> * <cast-expression>
                                | <multiplicative-expression> / <cast-expression>
                                | <multiplicative-expression> % <cast-expression>

  22.) <cast-expression> ::= <unary-expression>
                      | ( <type-name> ) <cast-expression>

  23.) <unary-expression> ::= <postfix-expression>
                      | ++ <unary-expression>
                      | -- <unary-expression>
                      | <unary-operator> <cast-expression>
                      | sizeof <unary-expression>
                      | sizeof <type-name>

  24.) <postfix-expression> ::= <primary-expression>
                        | <postfix-expression> [ <expression> ]
                        | <postfix-expression> ( {<assignment-expression>}* )
                        | <postfix-expression> . <identifier>
                        | <postfix-expression> -> <identifier>
                        | <postfix-expression> ++
                        | <postfix-expression> --

  25.) <primary-expression> ::= <identifier>
                        | <constant>
                        | <string>
                        | ( <expression> )

  26.) <constant> ::= <integer-constant>
              | <character-constant>
              | <floating-constant>

  27.) <expression> ::= <assignment-expression>
                | <expression> , <assignment-expression>

  28.) <assignment-expression> ::= <logical-or-expression>
                            | <unary-expression> <assignment-operator> <assignment-expression>

  29.) <assignment-operator> ::= =
                          | *=
                          | /=
                          | %=
                          | +=
                          | -=

  30.) <unary-operator> ::= *
                    | +
                    | -
                    | !

  31.) <type-name> ::= {<specifier-qualifier>}+ {<abstract-declarator>}?

  32.) <parameter-type-list> ::= <parameter-list>
                          | <parameter-list> , ...

  33.) <parameter-list> ::= <parameter-declaration>
                    | <parameter-list> , <parameter-declaration>

  34.) <parameter-declaration> ::= {<declaration-specifier>}+ <declarator>
                            | {<declaration-specifier>}+ <abstract-declarator>
                            | {<declaration-specifier>}+

  35.) <abstract-declarator> ::= <pointer>
                          | <pointer> <direct-abstract-declarator>
                          | <direct-abstract-declarator>

  36.) <direct-abstract-declarator> ::=  ( <abstract-declarator> )
                                | {<direct-abstract-declarator>}? [ {<constant-expression>}? ]
                                | {<direct-abstract-declarator>}? ( {<parameter-type-list>}? )

  37.) <declaration> ::=  {<declaration-specifier>}+ {<init-declarator>}* ;

  38.) <init-declarator> ::= <declarator>
                      | <declarator> = <initializer>

  39.) <initializer> ::= <assignment-expression>
                  | { <initializer-list> }
                  | { <initializer-list> , }

  40 .)<initializer-list> ::= <initializer>
                      | <initializer-list> , <initializer>

  41.) <compound-statement> ::= { {<declaration>}* {<statement>}* }

  42.) <statement> ::= <expression-statement>
                | <compound-statement>
                | <selection-statement>
                | <iteration-statement>
                | <jump-statement>

  43.) <expression-statement> ::= {<expression>}? ;

  44.) <selection-statement> ::= if ( <expression> ) <statement>
                          | if ( <expression> ) <statement> else <statement>

  45.) <iteration-statement> ::= while ( <expression> ) <statement>
                          | for ( {<expression>}? ; {<expression>}? ; {<expression>}? ) <statement>

  46.) <jump-statement> ::= continue ;
                    | break ;
                    | return {<expression>}? ;
